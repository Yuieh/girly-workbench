function esc(s){return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');}
function get(path,def){let o=db;for(const k of path){if(o==null||o[k]==null)return def;o=o[k];}return o;}
function set(path,val){let o=db;for(let i=0;i<path.length-1;i++){if(o[path[i]]==null)o[path[i]]={};o=o[path[i]];}o[path[path.length-1]]=val;save();}

/* ================== 通用列表组件 ================== */
function todoListHTML(path,items,opts={}){
  const p=JSON.stringify(path).replace(/"/g,'&quot;');
  let html=items.map(it=>`
    <div class="todo-item ${it.done?'done':''}">
      <input type="checkbox" class="chk" ${it.done?'checked':''} onchange="tToggle(${p},'${it.id}')">
      <div class="todo-body">
        <input class="todo-text" value="${esc(it.text)}" placeholder="写点什么…" onchange="tEdit(${p},'${it.id}','text',this.value)">
        <input class="todo-note" value="${esc(it.note||'')}" placeholder="备注小字…" onchange="tEdit(${p},'${it.id}','note',this.value)">
        <input class="todo-time" type="time" value="${esc(it.time||'')}" onchange="tEdit(${p},'${it.id}','time',this.value)">
      </div>
      <button class="del-btn" onclick="tDel(${p},'${it.id}')">✕</button>
    </div>`).join('');
  if(!items.length) html=`
    <div class="todo-item">
      <input type="checkbox" class="chk" disabled>
      <div class="todo-body">
        <input class="todo-text" placeholder="写点什么…" onchange="tNew(${p},'text',this.value)">
        <input class="todo-note" placeholder="备注小字…" onchange="tNew(${p},'note',this.value)">
        <input class="todo-time" type="time" onchange="tNew(${p},'time',this.value)">
      </div>
    </div>`;
  return html+(items.length?`<button class="add-btn" onclick="tAdd(${p})">＋ ${opts.addText||'添加一项'}</button>`:'');
}
function tNew(path,field,value){
  if(field==='text'&&!String(value).trim())return;
  const it={id:uid(),text:'',note:'',time:nowTime(),done:false};it[field]=value;
  tGetList(path).push(it);save();renderAll();
}
function calFoodNew(meal,k,v){
  if(k==='name'&&!String(v).trim())return;
  const f={id:uid(),name:'',grams:100,per100:0,kcal:0,auto:false,meal:meal||''};f[k]=v;
  if(k==='name'){const hit=Object.keys(FOOD_DB).find(n=>v.includes(n)||n.includes(v));if(hit&&v.trim()){f.per100=FOOD_DB[hit];f.auto=true;}}
  f.kcal=Math.round((+f.grams||0)*(+f.per100||0)/100);
  calData().foods.push(f);save();renderAll();
}
function tGetList(path){let l=get(path,null);if(!l){l=[];set(path,l);}return l;}
function tAdd(path){tGetList(path).push({id:uid(),text:'',note:'',time:nowTime(),done:false});save();renderAll();}
function tToggle(path,id){const l=tGetList(path);const it=l.find(x=>x.id===id);if(it){it.done=!it.done;save();renderAll();}}
function tEdit(path,id,k,v){const l=tGetList(path);const it=l.find(x=>x.id===id);if(it){it[k]=v;save();}}
function tDel(path,id){set(path,tGetList(path).filter(x=>x.id!==id));renderAll();}

/* 照片上传 */
function pickPhoto(cb){
  const inp=document.createElement('input');inp.type='file';inp.accept='image/*';
  inp.onchange=()=>{const f=inp.files[0];if(!f)return;const r=new FileReader();
    r.onload=()=>{window[cb](r.result);};r.readAsDataURL(f);};
  inp.click();
}

function tarotFullText(name,pos){
  const d=TAROT_DB[name];if(!d)return '';
  const rev=pos==='逆位';
  return `【${name} · ${rev?'逆位':'正位'}】
🏷 关键词：${d.kw}

🔮 牌意：${rev?d.down:d.up}

💕 感情：${rev?d.ld:d.lu}
💼 事业：${rev?d.wd:d.wu}
💰 财运：${rev?d.md:d.mu}`;
}

function tarotCardAdd(){tarotData().cards.push({id:uid(),name:'',pos:'正位',meaning:''});save();renderAll();}
function tarotNewCard(name){
  const n=(name||'').trim();
  if(!n)return;
  /* 模糊匹配：用户可能输入简写 */
  let match=n;
  if(!TAROT_DB[n]){
    const found=TAROT_NAMES.find(k=>k.includes(n)||n.includes(k));
    if(found)match=found;
  }
  const c={id:uid(),name:match,pos:'正位',meaning:''};
  if(TAROT_DB[match])c.meaning=tarotFullText(match,'正位');
  tarotData().cards.push(c);save();renderAll();
}
function tarotCardEdit(id,k,v){
  const c=tarotData().cards.find(x=>x.id===id);if(!c)return;
  c[k]=v;
  if(k==='name'&&c.name&&!TAROT_DB[c.name]){
    /* 模糊匹配：输「愚」→「愚人」 */
    const f=TAROT_NAMES.find(x=>x.includes(c.name)||c.name.includes(x));
    if(f)c.name=f;
  }
  if((k==='name'||k==='pos')&&c.name&&TAROT_DB[c.name]){
    c.meaning=tarotFullText(c.name,c.pos||'正位');
    save();renderAll();return;
  }
  save();/* 手改牌意时不重渲染，避免失焦跳动 */
}
function tarotResetMeaning(id){
  const c=tarotData().cards.find(x=>x.id===id);if(!c||!TAROT_DB[c.name])return;
  c.meaning=tarotFullText(c.name,c.pos||'正位');save();renderAll();
}
function tarotCardDel(id){const t=tarotData();t.cards=t.cards.filter(x=>x.id!==id);save();renderAll();}
function tarotAutoCombo(){
  try{
    const t=tarotData();const cs=(t.cards||[]).filter(c=>c.name&&c.name.trim());
    if(cs.length<2){alert('⚠️ 至少需要 2 张已选的牌才能生成组合牌意哦～\n当前已有 '+cs.length+' 张');return;}
    const SUIT=[['权杖','🔥 火 · 行动与热情'],['圣杯','💧 水 · 情感与关系'],['宝剑','🗡 风 · 思维与沟通'],['星币','🪙 土 · 物质与务实']];
    const lines=cs.map((c,i)=>{
      const d=TAROT_DB[c.name];const rev=c.pos==='逆位';
      if(!d)return `第${i+1}张：${c.name}${rev?'（逆位）':'（正位）'}\n→ (自定义牌，无字典数据)`;
      return `第${i+1}张：${c.name}${rev?'（逆位）':'（正位）'}\n🏷 ${d.kw}\n→ ${rev?d.down:d.up}\n   💕${rev?d.ld:d.lu}\n   💼${rev?d.wd:d.wu}\n   💰${rev?d.md:d.mu}`;
    });
    /* 元素分布分析 */
    const cnt={};cs.forEach(c=>{const s=SUIT.find(s=>c.name.startsWith(s[0]));cnt[s?s[0]:'大牌']=(cnt[s?s[0]:'大牌']||0)+1;});
    const majorN=cnt['大牌']||0;
    let el='';
    if(majorN>=2&&majorN>=cs.length/2)el=`\n⚡ 大阿尔卡纳占了 ${majorN}/${cs.length} 张——这件事是命运级别的课题，不是小打小闹，值得认真对待。`;
    const domi=SUIT.find(s=>(cnt[s[0]]||0)>=2&&(cnt[s[0]]||0)>=cs.length/2);
    if(domi)el+=`\n⚡ ${domi[0]}牌集中出现——${domi[1]}，这是当前的主旋律。`;
    const revN=cs.filter(c=>c.pos==='逆位').length;
    const revTip=revN===0?'全为正位，能量顺畅，可以大胆推进。'
      :revN===cs.length?'全为逆位，阻力明显，此刻宜内省、调整，先别硬推。'
      :`${revN} 张逆位 / ${cs.length-revN} 张正位，顺逆交错——有机会也有功课，边走边校准。`;
    t.combo=`🔮 「${cs.map(c=>c.name+(c.pos==='逆位'?'(逆)':'')).join(' + ')}」组合解读\n${'─'.repeat(30)}\n\n${lines.join('\n\n')}\n\n${'─'.repeat(30)}\n\n🌙 整体解读：\n这组牌像是一段从「${cs[0].name}」走向「${cs[cs.length-1].name}」的旅程。\n前者是你当下的处境与心境，后者指向事情发展的方向。\n\n⚖️ 能量走向：${revTip}${el}\n\n🎯 结合你的问题「${t.question||'未设置'}」细细体会，答案就在牌面之间✨`;
    save();renderAll();
    /* 确保 textarea 更新显示 */
    setTimeout(()=>{const el=document.getElementById('comboText');if(el)el.value=t.combo;},100);
  }catch(e){alert('生成出错：'+e.message);}
}

    const detail=d?`\n\n🏷 核心关键词：${d.kw}\n\n💕 感情面：${rev?d.ld:d.lu}\n💼 事业面：${rev?d.wd:d.wu}\n💰 财运面：${rev?d.md:d.mu}`:'';
    t.guide=`🌟 今日指导 · 以「${last.name}${rev?' 逆位':' 正位'}」为主轴\n${'─'.repeat(30)}\n\n${meaning}${detail}\n\n${'─'.repeat(30)}\n\n💫 行动建议：${advice}\n\n🎀 幸运提示：保持觉察，把牌面提示和「${t.question||'今日问题'}」对照，晚上回顾一下应验了哪些吧～`;
    save();renderAll();
    setTimeout(()=>{const el=document.getElementById('guideText');if(el)el.value=t.guide;},100);
  }catch(e){alert('生成出错：'+e.message);}
}

const MOOD_EMOJIS=['😍','😊','😐','😢','😠','😭','😴','✨'];
function treeList(k){if(!db.tree)db.tree={};if(!db.tree[k])db.tree[k]=[];return db.tree[k];}

/* --- 日常随笔（保持原样）--- */
function renderTreeEssay(){
  const list=treeList('essay');
  const cards=[...list].reverse().map(it=>`
    <div class="rec-card">
      <div class="rec-head"><span class="rec-date">🗓 ${esc(it.date)} ${esc(it.time||'')}</span><button class="del-btn" onclick="treeDel('essay','${it.id}')">✕</button></div>
      <textarea class="f-area" style="min-height:60px" placeholder="今天的碎碎念、小感悟…" onchange="treeEdit('essay','${it.id}','text',this.value)">${esc(it.text||'')}</textarea>
    </div>`).join('')||`<div class="rec-card">
      <div class="rec-head"><span class="rec-date">🗓 ${todayStr()} ${nowTime()}</span></div>
      <textarea class="f-area" style="min-height:60px" placeholder="今天的碎碎念、小感悟…" onchange="treeNew('essay',this.value)"></textarea>
    </div>`;
  return `<div class="card"><div class="card-title">✏️ 日常随笔 <span class="tag">私密 · 仅本机保存</span></div>
    ${list.length?`<button class="add-btn" style="margin-bottom:12px" onclick="treeAdd('essay')">＋ 写一条新的</button>`:''}${cards}</div>`;
}
function treeNew(k,v){
  if(!String(v).trim())return;
  treeList(k).push({id:uid(),date:todayStr(),time:nowTime(),text:v,mood:'',term:'短期',done:false});
  save();renderAll();
}

/* --- 情绪记录（截图3：emoji选择器+日记+记录按钮+历史本子）--- */
let _pickMood='';
function renderTreeMood(){
  const list=treeList('mood');
  /* 历史记录列表 */
  const history=[...list].reverse().map(it=>`
    <div class="rec-card">
      <div class="rec-head"><span class="rec-date">🗓 ${esc(it.date)} ${esc(it.time||'')} <span style="font-size:18px;">${esc(it.mood)}</span></span><button class="del-btn" onclick="treeDel('mood','${it.id}')">✕</button></div>
      <div style="white-space:pre-wrap;font-size:13px;color:var(--text);line-height:1.6;">${esc(it.text)||''}</div>
    </div>`).join('')||'<div class="empty-tip">还没有情绪记录 🌸</div>';
  return `
  <div class="card">
    <div class="card-title">🌈 今天心情如何？</div>
    <div class="mood-picker">${MOOD_EMOJIS.map(m=>`<div class="mood-emo ${_pickMood===m?'sel':''}" onclick="_pickMood='${m}';renderAll()">${m}</div>`).join('')}</div>
    <textarea id="moodDiaryInput" class="mood-diary" placeholder="写写心情日记..." oninput="_moodDraft=this.value"></textarea>
    <button class="add-btn-grad" onclick="saveMood()">记录心情 💖</button>
  </div>
  <div class="card">
    <div class="card-title">📖 心情日记录本</div>${history}
  </div>`;
}
function saveMood(){
  if(!_pickMood){alert('先选一个心情表情吧～');return;}
  const txt=document.getElementById('moodDiaryInput').document?document.getElementById('moodDiaryInput').value:'';
  treeList('mood').push({id:uid(),date:todayStr(),time:nowTime(),text:txt||'',mood:_pickMood});
  save();_pickMood='';renderAll();
}

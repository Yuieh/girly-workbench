function renderTreeWish(){
  const shortList=treeList('wish').filter(w=>w.term!=='长期');
  const longList=treeList('wish').filter(w=>w.term==='长期');
  const sp=JSON.stringify(['tree','wish']).replace(/"/g,'&quot;');
  const wishItem=(it)=>`<div class="wish-item ${it.done?'done':''}">
    <input type="checkbox" class="chk" style="width:20px;height:20px;border-radius:8px;" ${it.done?'checked':''} onchange="tToggle(${sp},'${it.id}')">
    <input class="wish-text" value="${esc(it.text)}" onchange="tEdit(${sp},'${it.id}','text',this.value)">
    <button class="del-btn" onclick="tDel(${sp},'${it.id}')">✕</button>
  </div>`;
  return `
  <div class="card">
    <div class="wish-group">
      <div class="wish-group-title">🌷 短期心愿</div>
      <div class="wish-hint">许个愿吧 ✨</div>
      <div class="wish-input-row">
        <input class="wish-input" id="wishShortIn" placeholder="想要实现...">
        <button class="wish-add-btn" onclick="addWish('short')">＋</button>
      </div>
      ${shortList.length?shortList.map(wishItem).join(''):''}
    </div>
    <div class="wish-group" style="margin-top:24px;">
      <div class="wish-group-title">🌟 长期心愿</div>
      <div class="wish-hint">许个愿吧 ✨</div>
      <div class="wish-input-row">
        <input class="wish-input" id="wishLongIn" placeholder="想要实现...">
        <button class="wish-add-btn" onclick="addWish('long')">＋</button>
      </div>
      ${longList.length?longList.map(wishItem).join(''):''}
    </div>
  </div>`;
}
function addWish(term){
  const inp=document.getElementById(term==='short'?'wishShortIn':'wishLongIn');
  const text=(inp&&inp.value)||'';
  treeList('wish').push({id:uid(),date:todayStr(),time:nowTime(),text,term:term==='long'?'长期':'短期',done:false,mood:'',note:''});
  save();if(inp)inp.value='';renderAll();
}

/* --- 灵感收集（截图5：大文本框+保存按钮+全部记录列表）--- */
function renderTreeIdea(){
  const list=treeList('idea');
  const records=[...list].reverse().map(it=>`
    <div class="rec-card">
      <div class="rec-head"><span class="rec-date">🗓 ${esc(it.date)} ${esc(it.time||'')}</span><button class="del-btn" onclick="treeDel('idea','${it.id}')">✕</button></div>
      <div style="white-space:pre-wrap;font-size:13px;color:var(--text);line-height:1.6;">${esc(it.text)||''}</div>
    </div>`).join('')||'<div class="empty-tip">空空的，等你来写 🎀</div>';
  return `
  <div class="card">
    <div class="card-title">💡 突发想法 · 文案 · 创意素材</div>
    <textarea id="ideaInput" class="idea-textarea" placeholder="灵感闪现！快记下来…"></textarea>
    <button class="add-btn-grad" onclick="saveIdea()">保存 📌</button>
  </div>
  <div class="card">
    <div class="idea-list-title">📦 全部记录</div>${records}
  </div>`;
}
function saveIdea(){
  const inp=document.getElementById('ideaInput');
  const text=(inp&&inp.value)||'';
  if(!text.trim())return;
  treeList('idea').push({id:uid(),date:todayStr(),time:nowTime(),text,mood:'',term:'',done:false,note:''});
  save();if(inp)inp.value='';renderAll();
}

function treeAdd(k){treeList(k).push({id:uid(),date:todayStr(),time:nowTime(),text:'',mood:k==='mood'?'🥰':'',term:'短期',done:false});save();renderAll();}
function treeEdit(k,id,f,v){const it=treeList(k).find(x=>x.id===id);if(it){it[f]=v;save();}}
function treeDel(k,id){db.tree[k]=treeList(k).filter(x=>x.id!==id);save();renderAll();}

function renderMatchCards(items){
  if(!items||!items.length)return '<div class="empty-tip">没有匹配到菜品呢，换个食材试试 💕</div>';
  return `<div style="margin-top:8px;">
    <div style="font-size:14px;font-weight:700;color:var(--red);margin-bottom:8px;">✅ 食材齐全可直接做（${items.length}道）</div>
    ${items.map((it,i)=>`
    <div class="rec-card" style="border-left:4px solid ${i%2===0?'var(--pink-deep)':'var(--blue-deep)'}">
      <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:4px;">

        <div style="display:flex;gap:6px;">
          <span class="pill" style="background:#FFF0E5;color:#E89230;font-size:10px;">${it.diff||'简单'}</span>
          <span style="color:#7BC496;font-weight:700;font-size:13px;">约${it.cal}kcal</span>
        </div>
      </div>
      <div style="font-size:11px;color:var(--sub);margin-bottom:4px;">主料：${esc(it.ingredients||'')}</div>
      <div style="font-size:12px;color:var(--text);line-height:1.6;background:rgba(189,224,254,.12);padding:8px;border-radius:10px;">${esc(it.steps)}</div>

function matchRecipes(){
  const inp=document.getElementById('recipeIngInput');
  const raw=(inp&&inp.value)||'';
  if(!raw.trim()){alert('先输入食材哦～');return;}
  const ings=raw.split(/[,，、\s]+/).map(s=>s.trim()).filter(Boolean);
  const rd=recipeData();rd.lastIngs=raw;
  /* 查找匹配的菜品（去重） */
  const foundMap=new Map();
  for(const ing of ings){
    const list=RECIPE_DB[ing]||[];
    for(const dish of list){
      if(!foundMap.has(dish.name))foundMap.set(dish.name,dish);
    }
  }
  const results=[...foundMap.values()].slice(0,4);
  rd.lastMatch=results;save();
  document.getElementById('recipeMatchResult').innerHTML=renderMatchCards(results);
}

/* 长按拖拽排序（移动端 touch） */
function setupPackingDrag(){
  const root=document.getElementById('content');
  if(!root||root._pkDragInited)return;
  root._pkDragInited=true;
  let pressTimer=null,dragging=false,el=null;
  const isCtrl=t=>t&&(t.closest('input')||t.closest('button')||t.closest('label'));
  root.addEventListener('touchstart',e=>{
    const item=e.target.closest('.packing-item');
    if(!item||isCtrl(e.target))return;
    el=item;
    pressTimer=setTimeout(()=>{dragging=true;el.classList.add('dragging');if(navigator.vibrate)navigator.vibrate(40);},350);
  },{passive:true});
  root.addEventListener('touchmove',e=>{
    if(!dragging||!el)return;
    e.preventDefault();
    const y=e.touches[0].clientY,list=el.parentNode;
    let ref=null;
    list.querySelectorAll('.packing-item:not(.dragging)').forEach(o=>{
      const r=o.getBoundingClientRect();
      if(y<r.top+r.height/2&&!ref)ref=o;
    });
    if(ref)list.insertBefore(el,ref);else list.appendChild(el);
  },{passive:false});
  const end=()=>{
    if(pressTimer)clearTimeout(pressTimer);
    if(!dragging){el=null;return;}
    dragging=false;
    if(el)el.classList.remove('dragging');
    const ids=[...root.querySelectorAll('.packing-item')].map(x=>x.getAttribute('data-id'));
    el=null;reorderPacking(ids);
  };
  root.addEventListener('touchend',end);
  root.addEventListener('touchcancel',end);
}
